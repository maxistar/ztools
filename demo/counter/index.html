import { signal, createTags, mount } from "../../ztools.js";

const [div, span, button] = createTags("div", "span", "button");

function App() {
  const count = signal(0);

  return div(
    { className: "counter-display" },
    span("Count: "),
    span({ id: "count-value" }, () => count())
  ),
  div(
    { className: "counter-controls" },
    button({ id: "decrement", onClick: () => count.set(count() - 1) }, "-"),
    button({ id: "increment", onClick: () => count.set(count() + 1) }, "+")
  );
}

mount(App, document.getElementById('app-container'));
